# Web UI Implementation Plan (Sub-project 3 of 3)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the React-based operator console from §17 of the design spec — Dashboard, Ontology Browser, Graph View, Diff/Promote, and Ingestion (upload + jobs list + job detail). Admin views (tokens / connections / audit log) stay MCP-only in V1.

**Architecture:** React 18 + Vite + TypeScript + TailwindCSS + shadcn/ui primitives. Routing via react-router data-router. Server state via TanStack Query (built-in interval refetch for ingestion polling). Forms via react-hook-form + zod. Graph viz via `react-flow` + `@dagrejs/dagre`. Auth via session cookie minted from a paste-token login (the `/admin/login` endpoint added in Plan #1). Built bundle is served by FastAPI as static files; dev runs Vite on port 5173 proxying API requests to the backend on 8080.

**Tech Stack:** Plan #1 + plan #2 deps for the backend. UI side: `vite`, `react`, `react-dom`, `react-router-dom`, `@tanstack/react-query`, `tailwindcss`, `@radix-ui/*` (via shadcn/ui), `react-hook-form`, `zod`, `reactflow`, `@dagrejs/dagre`, `@axe-core/playwright` (tests), `@playwright/test`. Reference: spec §17.

**Plan series:** Plan 3/3. Plans #1 and #2 must be implemented first; this plan adds a sibling `ui/` directory and modifies `docker/Dockerfile` to bundle the built UI into the same image.

```ts
// ui/src/api/client.ts
export class ApiError extends Error {
  constructor(public status: number, public code?: string, public details?: unknown) {
    super(`${status} ${code ?? ""}`.trim());
  }
}

export async function api<T = unknown>(path: string, init: RequestInit = {}): Promise<T> {
  const r = await fetch(path, {
    ...init,
    credentials: "include",
    headers: {
      "Content-Type": "application/json",
      ...(init.headers ?? {}),
    },
  });
  if (!r.ok) {
    let body: any = null;
    try { body = await r.json(); } catch { /* */ }
    throw new ApiError(r.status, body?.detail?.code ?? body?.code, body?.detail ?? body);
  }
  return r.json();
}
```

- [x] **Step 6: Implement session hook + route guard**

```ts
// ui/src/auth/session.ts
import { useQuery } from "@tanstack/react-query";
import { api, ApiError } from "@/api/client";

export interface Whoami { token_label: string; scope: "read" | "editor" | "admin"; server_version: string }

export function useSession() {
  return useQuery<Whoami | null>({
    queryKey: ["whoami"],
    queryFn: async () => {
      try {
        return await api<Whoami>("/api/whoami");
      } catch (e) {
        if (e instanceof ApiError && e.status === 401) return null;
        throw e;
      }
    },
    retry: false,
    staleTime: 30_000,
  });
}
```

```tsx
import { Navigate } from "react-router-dom";

export function RequireSession({ children }: { children: React.ReactNode }) {
  const q = useSession();
  if (q.isLoading) return <div className="p-8 text-zinc-400">Loading…</div>;
  if (!q.data) return <Navigate to="/login" replace />;
  return <>{children}</>;
}
```

```tsx
// ui/src/auth/login.tsx
import { useState } from "react";
import { api, ApiError } from "@/api/client";

export function Login({ onSuccess }: { onSuccess: () => void }) {
  const [token, setToken] = useState("");
  const [error, setError] = useState<string | null>(null);
  return (
    <form
      onSubmit={async (e) => {
        e.preventDefault();
        setError(null);
        try {
          await api("/admin/login", { method: "POST", body: JSON.stringify({ token }) });
          onSuccess();
        } catch (err) {
          if (err instanceof ApiError && err.status === 401) {
            setError("Invalid or revoked token.");
          } else {
            setError("Login failed.");
          }
        }
      }}
      className="mx-auto mt-32 w-96 space-y-4 rounded-lg border border-zinc-800 bg-zinc-900 p-6"
    >
      <h1 className="text-lg font-semibold">Sign in</h1>
      <label className="block text-sm">
        Token
        <input
          aria-label="token"
          value={token}
          onChange={(e) => setToken(e.target.value)}
          className="mt-1 w-full rounded border border-zinc-700 bg-zinc-950 p-2"
          autoFocus
        />
      </label>
      {error && <div role="alert" className="text-sm text-red-400">{error}</div>}
      <button className="w-full rounded bg-emerald-600 p-2 text-sm font-semibold">
        Sign in
      </button>
    </form>
  );
}

export default Login;
```

```ts
// ui/src/views/graph/build_graph.ts
import { Edge, Node } from "reactflow";

export interface OntologyForGraph {
  shared_property_types: Record<string, any>;
  interface_types: Record<string, any>;
  object_types: Record<string, any>;
  link_types: Record<string, any>;
  action_types: Record<string, any>;
}

const KIND_OF: Record<string, string> = {
  shprop: "SharedPropertyType",
  iface: "InterfaceType",
  obj: "ObjectType",
  link: "LinkType",
  action: "ActionType",
};

function kindFromRid(rid: string): string {
  const m = rid.match(/^ri\.([^.]+)\./);
  return m ? KIND_OF[m[1]] ?? m[1] : "?";
}

function pushEdge(out: Edge[], source: string, target: string, kind: string) {
  out.push({ id: `${kind}:${source}->${target}`, source, target, data: { kind }, type: "default" });
}

export function buildGraph(reg: OntologyForGraph): { nodes: Node[]; edges: Edge[] } {
  const nodes: Node[] = [];
  const edges: Edge[] = [];
  const all = [
    ...Object.values(reg.shared_property_types),
    ...Object.values(reg.interface_types),
    ...Object.values(reg.object_types),
    ...Object.values(reg.link_types),
    ...Object.values(reg.action_types),
  ];
  for (const e of all) {
    nodes.push({
      id: e.rid,
      data: { label: e.api_name || e.rid, kind: kindFromRid(e.rid) },
      position: { x: 0, y: 0 },
      type: "default",
    });
  }
  for (const obj of Object.values(reg.object_types)) {
    for (const ifaceRid of obj.implements_interface_type_rids ?? []) {
      pushEdge(edges, obj.rid, ifaceRid, "IMPLEMENTS");
    }
  }
  for (const lt of Object.values(reg.link_types)) {
    if (lt.source_object_type_rid) pushEdge(edges, lt.source_object_type_rid, lt.rid, "CONNECTS");
    if (lt.target_object_type_rid) pushEdge(edges, lt.rid, lt.target_object_type_rid, "CONNECTS");
    for (const ifaceRid of lt.implements_interface_type_rids ?? []) {
      pushEdge(edges, lt.rid, ifaceRid, "IMPLEMENTS");
    }
  }
  for (const iface of Object.values(reg.interface_types)) {
    for (const parent of iface.extends_interface_type_rids ?? []) {
      pushEdge(edges, iface.rid, parent, "EXTENDS");
    }
    for (const sp of iface.required_shared_property_type_rids ?? []) {
      pushEdge(edges, iface.rid, sp, "REQUIRES");
    }
  }
  for (const obj of Object.values(reg.object_types)) {
    for (const pt of Object.values(obj.property_types ?? {}) as any[]) {
      if (pt.inherit_from_shared_property_type_rid) {
        pushEdge(edges, pt.rid, pt.inherit_from_shared_property_type_rid, "BASED_ON");
      }
    }
  }
  for (const action of Object.values(reg.action_types)) {
    for (const p of action.parameters ?? []) {
      const t = p.derived_from_object_type_rid || p.derived_from_link_type_rid || p.derived_from_interface_type_rid;
      if (t) pushEdge(edges, action.rid, t, "OPERATES_ON");
    }
  }
  return { nodes, edges };
}
```

```tsx
import { useMemo } from "react";
import { useParams } from "react-router-dom";
import { useQuery } from "@tanstack/react-query";
import ReactFlow, { Background, Controls, MiniMap } from "reactflow";
import dagre from "@dagrejs/dagre";
import "reactflow/dist/style.css";
import { fetchFull } from "@/api/ontology";
import { buildGraph } from "./build_graph";

function layout(nodes: any[], edges: any[]) {
  const g = new dagre.graphlib.Graph();
  g.setGraph({ rankdir: "LR", nodesep: 30, ranksep: 80 });
  g.setDefaultEdgeLabel(() => ({}));
  for (const n of nodes) g.setNode(n.id, { width: 180, height: 40 });
  for (const e of edges) g.setEdge(e.source, e.target);
  dagre.layout(g);
  return nodes.map((n) => {
    const p = g.node(n.id);
    return { ...n, position: { x: p.x - 90, y: p.y - 20 } };
  });
}

export default function GraphView() {
  const { env = "staging" } = useParams();
  const q = useQuery({ queryKey: ["full", env], queryFn: () => fetchFull(env as any) });
  const { nodes, edges } = useMemo(() => {
    if (!q.data) return { nodes: [], edges: [] };
    const g = buildGraph(q.data as any);
    return { nodes: layout(g.nodes, g.edges), edges: g.edges };
  }, [q.data]);
  return (
    <div className="h-[calc(100vh-0px)]">
      <ReactFlow nodes={nodes} edges={edges} fitView>
        <Background />
        <Controls />
        <MiniMap />
      </ReactFlow>
    </div>
  );
```

```ts
  await expect(page.getByText("material", { exact: false })).toBeVisible();
  await page.getByText("material", { exact: false }).click();
  await expect(page.getByText(/"rid": "ri.obj/)).toBeVisible();

test("graph view renders without overlapping nodes for fixture", async ({ page }) => {
  await page.goto("/graph/production");
  await page.waitForSelector(".react-flow__node");
  const nodes = await page.locator(".react-flow__node").all();
  const boxes = await Promise.all(nodes.map(n => n.boundingBox()));
  // Sanity: every node has a position; none overlap by their full extent
  for (const b of boxes) expect(b).not.toBeNull();
  for (let i = 0; i < boxes.length; i++) {
    for (let j = i + 1; j < boxes.length; j++) {
      const a = boxes[i]!, b = boxes[j]!;
      const overlap = !(a.x + a.width <= b.x || b.x + b.width <= a.x ||
                        a.y + a.height <= b.y || b.y + b.height <= a.y);
      expect(overlap).toBe(false);
    }
  }
});
```
